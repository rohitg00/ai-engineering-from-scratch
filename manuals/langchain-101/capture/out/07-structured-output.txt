# Structured output: each strategy

## a bare schema on a model with no profile: AutoStrategy picks ToolStrategy

structured_response:
{
  "value": 5,
  "unit": "none"
}

structured_response type: Answer

[
  {
    "type": "AIMessage",
    "content": "",
    "id": "lc_run--<uuid-1>-0",
    "tool_calls": [
      {
        "name": "Answer",
        "args": {
          "value": 5
        },
        "id": "call_1",
        "type": "tool_call"
      }
    ]
  },
  {
    "type": "ToolMessage",
    "content": "Returning structured response: value=5 unit='none'",
    "name": "Answer",
    "id": "<uuid-2>",
    "tool_call_id": "call_1",
    "status": "success"
  }
]

model calls seen by the fake:
[
  {
    "model": "no-profile",
    "call": 1,
    "messages": [
      "human: What is 2 + 3?"
    ],
    "bound": {
      "tools": [
        "add",
        "Answer"
      ],
      "tool_choice": "any",
      "kwargs": []
    }
  }
]
