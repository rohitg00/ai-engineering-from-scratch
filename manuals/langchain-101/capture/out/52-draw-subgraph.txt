# Drawing a parent graph with a subgraph, with and without xray

## get_graph(): nodes

[
  "__start__",
  "prepare",
  "inner",
  "finish",
  "__end__"
]

## get_graph(): edges

[
  "__start__ -> prepare",
  "inner -> finish",
  "prepare -> inner",
  "finish -> __end__"
]

## get_graph().draw_mermaid()

---
config:
  flowchart:
    curve: linear
---
graph TD;
	__start__([<p>__start__</p>]):::first
	prepare(prepare)
	inner(inner)
	finish(finish)
	__end__([<p>__end__</p>]):::last
	__start__ --> prepare;
	inner --> finish;
	prepare --> inner;
	finish --> __end__;
	classDef default fill:#f2f0ff,line-height:1.2
	classDef first fill-opacity:0
	classDef last fill:#bfb6fc


## get_graph().draw_ascii()

draw_ascii():

## get_graph(xray=True): nodes

[
  "__start__",
  "prepare",
  "finish",
  "__end__",
  "inner:shout",
  "inner:ask_ok"
]

## get_graph(xray=True): edges

[
  "__start__ -> prepare",
  "inner:ask_ok -> finish",
  "prepare -> inner:shout",
  "finish -> __end__",
  "inner:shout -> inner:ask_ok"
]

## get_graph(xray=True).draw_mermaid()

---
config:
  flowchart:
    curve: linear
---
graph TD;
	__start__([<p>__start__</p>]):::first
	prepare(prepare)
	finish(finish)
	__end__([<p>__end__</p>]):::last
	__start__ --> prepare;
	inner\3aask_ok --> finish;
	prepare --> inner\3ashout;
	finish --> __end__;
	subgraph inner
	inner\3ashout(shout)
	inner\3aask_ok(ask_ok)
	inner\3ashout --> inner\3aask_ok;
	end
	classDef default fill:#f2f0ff,line-height:1.2
	classDef first fill-opacity:0
	classDef last fill:#bfb6fc
