	__start__([<p>__start__</p>]):::first
	prepare(prepare)
	finish(finish)
	__end__([<p>__end__</p>]):::last
	__start__ --> prepare;
	inner\3aask_ok --> finish;
	prepare --> inner\3ashout;
	finish --> __end__;
	subgraph inner
	inner\3ashout(shout)
	inner\3aask_ok(ask_ok)
	inner\3ashout --> inner\3aask_ok;
	end
