# The tools stream mode

## stream_mode='tools': tool lifecycle events and output deltas

chunk 0:
{
  "event": "tool-started",
  "tool_call_id": "call_1",
  "tool_name": "slow_sum",
  "input": {
    "a": 2,
    "b": 3
  }
}

chunk 1:
{
  "event": "tool-output-delta",
  "tool_call_id": "call_1",
  "delta": {
    "partial": 2
  }
}

chunk 2:
{
  "event": "tool-output-delta",
  "tool_call_id": "call_1",
  "delta": {
    "partial": 5
  }
}

chunk 3:
{
  "event": "tool-finished",
  "tool_call_id": "call_1",
  "output": {
    "type": "ToolMessage",
    "content": "5",
    "name": "slow_sum",
    "id": "<uuid-1>",
    "tool_call_id": "call_1",
    "status": "success"
  }
}
